cmake_minimum_required(VERSION 3.20)
project(AuraRng VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(aura_core src/core/aura.cpp)
target_include_directories(aura_core PUBLIC src)
if(MSVC)
  target_compile_options(aura_core PUBLIC /W4 /permissive- /utf-8 /EHsc)
  set_property(TARGET aura_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(aura_core PUBLIC -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(aura_tests tests/core_tests.cpp)
  target_link_libraries(aura_tests PRIVATE aura_core)
  if(MSVC)
    set_property(TARGET aura_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  add_test(NAME core COMMAND aura_tests "${CMAKE_CURRENT_SOURCE_DIR}/data/auras.csv")
endif()
if(WIN32)
  add_executable(AuraRng WIN32 src/platform/main.cpp src/render/renderer.cpp src/platform/app.manifest)
  target_compile_definitions(AuraRng PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(AuraRng PRIVATE aura_core d3d11 dxgi d3dcompiler d2d1 dwrite windowscodecs shell32 ole32 user32 gdi32)
  set_property(TARGET AuraRng PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  if(MSVC)
    target_link_options(AuraRng PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/platform/app.manifest)
  endif()
  add_custom_command(TARGET AuraRng POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/data" "$<TARGET_FILE_DIR:AuraRng>/data")
  install(TARGETS AuraRng RUNTIME DESTINATION .)
  install(DIRECTORY data DESTINATION .)
  install(DIRECTORY docs DESTINATION .)
  install(FILES README.md DESTINATION .)
endif()
