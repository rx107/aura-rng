cmake_minimum_required(VERSION 3.20)
project(StarsRng VERSION 0.1.18 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(aura_core src/core/aura.cpp src/core/presentation.cpp src/core/animation.cpp)
target_include_directories(aura_core PUBLIC src)
if(MSVC)
  target_compile_options(aura_core PUBLIC /W4 /permissive- /utf-8 /EHsc)
  set_property(TARGET aura_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(aura_core PUBLIC -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(aura_tests tests/core_tests.cpp)
  target_link_libraries(aura_tests PRIVATE aura_core)
  if(MSVC)
    set_property(TARGET aura_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  add_test(NAME core COMMAND aura_tests "${CMAKE_CURRENT_SOURCE_DIR}/data/auras.csv")
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME multiplayer_server COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/multiplayer_server_tests.py")
    set_tests_properties(multiplayer_server PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" TIMEOUT 90)
  endif()
endif()
if(WIN32)
  if(BUILD_TESTING)
    add_executable(discord_tests tests/discord_tests.cpp src/platform/discord.cpp)
    target_include_directories(discord_tests PRIVATE src third_party)
    target_link_libraries(discord_tests PRIVATE advapi32)
    target_compile_definitions(discord_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MSVC)
      target_compile_options(discord_tests PRIVATE /W4 /utf-8 /EHsc)
      set_property(TARGET discord_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    add_test(NAME discord_ipc COMMAND discord_tests)
    add_executable(multiplayer_client_tests tests/multiplayer_client_tests.cpp src/platform/multiplayer.cpp src/platform/profile_photo.cpp)
    target_include_directories(multiplayer_client_tests PRIVATE src third_party)
    target_link_libraries(multiplayer_client_tests PRIVATE winhttp bcrypt crypt32 windowscodecs ole32)
    target_compile_definitions(multiplayer_client_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MSVC)
      target_compile_options(multiplayer_client_tests PRIVATE /W4 /utf-8 /EHsc)
      set_property(TARGET multiplayer_client_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    if(Python3_Interpreter_FOUND)
      add_test(NAME multiplayer_client COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/run_multiplayer_client_test.py"
        --executable $<TARGET_FILE:multiplayer_client_tests> --scratch-parent "${CMAKE_CURRENT_BINARY_DIR}/multiplayer-test-data")
      set_tests_properties(multiplayer_client PROPERTIES TIMEOUT 60)
    endif()
  endif()
  set(IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui")
  add_library(dear_imgui STATIC ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp
    ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp
    ${IMGUI_DIR}/backends/imgui_impl_win32.cpp ${IMGUI_DIR}/backends/imgui_impl_dx11.cpp)
  target_include_directories(dear_imgui PUBLIC ${IMGUI_DIR})
  target_compile_definitions(dear_imgui PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    target_compile_options(dear_imgui PRIVATE /utf-8 /EHsc)
    set_property(TARGET dear_imgui PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  add_executable(AuraRng WIN32 src/platform/main.cpp src/platform/audio.cpp src/platform/discord.cpp src/platform/multiplayer.cpp src/platform/profile_photo.cpp src/render/renderer.cpp src/render/effects.cpp src/render/pixelation.cpp src/render/equinox.cpp src/render/cutscene.cpp src/render/breakthrough.cpp src/render/nanally.cpp src/render/illusionary.cpp src/render/desert.cpp src/render/frutiger.cpp src/render/redjohn.cpp src/render/redjohn_cutscene.cpp src/render/multiplayer.cpp src/render/multiplayer_ui.cpp src/render/vrm.cpp src/render/ui.cpp src/platform/app.manifest)
  target_include_directories(AuraRng PRIVATE third_party)
  set_target_properties(AuraRng PROPERTIES OUTPUT_NAME StarRng)
  target_compile_definitions(AuraRng PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(AuraRng PRIVATE aura_core dear_imgui d3d11 dxgi d3dcompiler d2d1 dwrite windowscodecs shell32 ole32 user32 gdi32 dwmapi imm32 mfplat mfreadwrite mfuuid xaudio2 winhttp bcrypt crypt32)
  set_property(TARGET AuraRng PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  if(MSVC)
    target_link_options(AuraRng PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/platform/app.manifest)
  endif()
  add_custom_command(TARGET AuraRng POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/data" "$<TARGET_FILE_DIR:AuraRng>/data")
  install(TARGETS AuraRng RUNTIME DESTINATION .)
  install(DIRECTORY data DESTINATION .)
  install(DIRECTORY docs DESTINATION . PATTERN "*.png" EXCLUDE)
  install(DIRECTORY server DESTINATION . PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
  install(FILES README.md Start-with-local-assets.cmd Start-local-server.cmd DESTINATION .)
  install(FILES third_party/imgui/LICENSE.txt DESTINATION licenses RENAME Dear-ImGui-LICENSE.txt)
  install(FILES third_party/README.md DESTINATION third_party)
  install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION licenses RENAME nlohmann-json-LICENSE.txt)
  install(FILES third_party/cgltf/LICENSE DESTINATION licenses RENAME cgltf-LICENSE.txt)
  install(FILES third_party/stb/LICENSE.txt DESTINATION licenses RENAME stb-vorbis-LICENSE.txt)
endif()
